25일차 · 공개 snapshot을 MongoDB context에 멱등 반영한다
24일차 공개 player-snapshot/v1 파일에서 정규화·Mongo 복사본·재개 위치·본인 조회를 작성한다. 완성 binlog consumer가 있다고 전제하지 않는다. checksum과 다음 행으로 재개하고 삭제는 명시 teaching-fixture로만 학습한다. 광고주는 ad_server 웹, 게임 유저는 기존 Pygame·게임 중계를 이어 쓴다. 광고는 게임 ORM과 계정 비밀을 import하지 않는다. 현재 세 패키지와 표준 라이브러리만 사용한다.

입력 읽기 helper는 수신 파일과 계약을 같은 작업 폴더 기준으로 읽고, 실제 파일의 행 수·공개 ID·수집 시각·SHA-256을 계약과 대조한다.
artifact_id
는 안정 식별자이고
source_path
는 실제 수신 파일의 경로다. 검사·실행 중 원문을 동결해 둔다. helper는 검사 후 읽은 bytes의 SHA도 다시 대조하며 다른 프로세스의 변경을 막는 잠금을 제공하지 않는다. 아래 제공부를 모듈의 import 다음에 저장한다.
def read_snapshot_package(source, contract_path):
    """24일차의 동결된 원문·수신 계약을 소비하기 직전에 대조한다."""
    source = Path(source)
    contract_path = Path(contract_path)
    contract = json.loads(contract_path.read_text(encoding="utf-8"))
    if not isinstance(contract, dict):
        raise ValueError("계약은 JSON 객체여야 합니다.")
    inspected = inspect_snapshot(source)
    expected = {
        **inspected,
        "artifact_id": "data/exports/player-cdc.ndjson",
        "artifact_version": "v2",
        "collector": "game.export_player_snapshot",
        "binlog_cdc_status": "not-configured",
        "position_kind": "file-line",
        "kafka_position": None,
        "public_fields": list(PUBLIC_FIELDS),
    }
    for field, value in expected.items():
        if (field not in contract
                or json.dumps(contract[field], sort_keys=True)
                != json.dumps(value, sort_keys=True)):
            raise ValueError(f"계약과 원본이 다릅니다: {field}")
    parse_utc(contract.get("created_at"))
    declared_path = contract.get("source_path")
    if not isinstance(declared_path, str) or not declared_path:
        raise ValueError("계약 source_path가 없습니다.")
    if Path(declared_path).resolve() != source.resolve():
        raise ValueError("계약 source_path와 --source가 다릅니다.")
    # 같은 bytes에서 얻은 행·SHA를 사용한다. 검사 후 변경도 거절한다.
    rows, checksum = _read_snapshot(source)
    if checksum != inspected["sha256"]:
        raise ValueError("검사 후 원본이 변경됐습니다.")
    return rows, inspected

def normalize_snapshot(row, media_id="village-game"):
    if not isinstance(row, dict) or set(row) != SNAPSHOT_FIELDS:
        raise ValueError("공개 snapshot의 정확한 8개 필드가 필요합니다.")
    if row["schema_version"] != SCHEMA_VERSION or row["source_kind"] != SOURCE_KIND:
        raise ValueError("player-snapshot/v1 공개 입력이 필요합니다.")
    if not isinstance(media_id, str) or not media_id.strip():
        raise ValueError("media_id는 비어 있지 않은 문자열이어야 합니다.")
    _check_public_state({field: row[field] for field in PUBLIC_FIELDS})
    parse_utc(row["captured_at"])
    subject_id = str(row["id"])
    updated_at = parse_utc(row["updated_at"]).isoformat()
    context = {"player_id": subject_id, "room_id": row["room_id"], "coins": row["coins"],
               "version": row["version"], "updated_at": updated_at}
    change = {"schema_version": "player-context-change/v1",
              "source_kind": SOURCE_KIND, "media_id": media_id,
              "subject_id": subject_id, "op": "upsert", "version": row["version"],
              "event_time": updated_at, "context": context}
    canonical = json.dumps(change, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    change["change_id"] = hashlib.sha256(canonical.encode("utf-8")).hexdigest()
    return change

def normalize_snapshot_file(source, contract_path, output, media_id="village-game"):
    if not isinstance(media_id, str) or not media_id.strip():
        raise ValueError("media_id는 비어 있지 않은 문자열이어야 합니다.")
    target = Path(output)
    if target.resolve() in {Path(source).resolve(), Path(contract_path).resolve()}:
        raise ValueError("출력은 원본·계약과 다른 경로여야 합니다.")
    rows, inspected = read_snapshot_package(source, contract_path)
    changes = [normalize_snapshot(row, media_id=media_id) for row in rows]
    unique_ids = len({change["change_id"] for change in changes})
    if unique_ids != len(changes):
        raise ValueError("정규화 change_id가 중복됐습니다.")
    target.parent.mkdir(parents=True, exist_ok=True)
    # 모두 변환한 뒤 임시 파일에 쓰고, 완성된 파일만 최종 경로로 옮긴다.

        staged = Path(folder) / "changes.ndjson"
        checksum = write_ndjson(staged, changes)
        os.replace(staged, target)
    return {"path": str(target), "rows": len(changes), "sha256": checksum,
            "schema_version": "player-context-change/v1", "source_kind": SOURCE_KIND,

def context_subject_key(media_id, subject_id):
    """매체와 공개 대상을 구분하는 안정 키를 만든다."""
    for name, value in (("media_id", media_id), ("subject_id", subject_id)):
        if not isinstance(value, str) or not value.strip() or ":" in value:
            raise ValueError(f"{name}는 콜론 없는 비어 있지 않은 문자열이어야 합니다.")
    return media_id + ":" + subject_id


def _check_context_change(change):
    """정규화 계약을 대조한다. 파일이나 MongoDB를 변경하지 않는다."""
    fields = {"schema_version", "source_kind", "media_id", "subject_id", "op",
              "version", "event_time", "context", "change_id"}
    if not isinstance(change, dict) or set(change) != fields:
        raise ValueError("context 변경은 정해진 9개 필드만 받습니다.")
    if change["schema_version"] != "player-context-change/v1":
        raise ValueError("player-context-change/v1 입력이 필요합니다.")
    if change["source_kind"] not in {"player-snapshot", "teaching-fixture"}:
        raise ValueError("source_kind는 player-snapshot 또는 teaching-fixture입니다.")
    if change["op"] not in {"upsert", "delete"}:
        raise ValueError("op는 upsert 또는 delete입니다.")
    context_subject_key(change["media_id"], change["subject_id"])
    subject = change["subject_id"]
    if change["source_kind"] == "player-snapshot":
        if not subject.isascii() or not subject.isdecimal() or int(subject) <= 0 or str(int(subject)) != subject:
            raise ValueError("player-snapshot의 subject_id는 양의 공개 ID를 그대로 바꾼 문자열입니다.")
        if change["op"] != "upsert":
            raise ValueError("player-snapshot은 upsert만 허용합니다.")
    elif not subject.startswith("fixture-") or subject == "fixture-":
        raise ValueError("teaching-fixture는 fixture-로 시작하는 별도 subject가 필요합니다.")
    if type(change["version"]) is not int or change["version"] < 0:
        raise ValueError("version은 0 이상 정수입니다.")
    instant = parse_utc(change["event_time"])
    if instant.isoformat() != change["event_time"]:
        raise ValueError("event_time은 UTC로 정규화한 ISO 시각이어야 합니다.")
    context = change["context"]
    if not isinstance(context, dict):
        raise ValueError("context는 사전이어야 합니다.")
    if change["op"] == "delete":
        if context:
            raise ValueError("delete의 context는 빈 사전이어야 합니다.")
    else:
        if set(context) != {"player_id", "room_id", "coins", "version", "updated_at"}:
            raise ValueError("upsert context는 정해진 5개 공개 필드만 받습니다.")
        if context["player_id"] != subject:
            raise ValueError("context.player_id와 subject_id가 다릅니다.")
        if type(context["version"]) is not int or context["version"] != change["version"]:
            raise ValueError("context.version과 version이 다릅니다.")
        if not isinstance(context["room_id"], str) or type(context["coins"]) is not int:
            raise ValueError("room_id는 문자열, coins는 정수여야 합니다.")
        parse_utc(context["updated_at"])
        if context["updated_at"] != change["event_time"]:
            raise ValueError("context.updated_at과 event_time이 다릅니다.")
    body = {name: value for name, value in change.items() if name != "change_id"}
    canonical = json.dumps(body, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    expected_id = hashlib.sha256(canonical.encode("utf-8")).hexdigest()
    if change["change_id"] != expected_id:
        raise ValueError("change_id와 정규화 내용이 다릅니다.")

def make_context_document(change):
    """검증한 변경 한 행을 저장할 문서로 바꾼다. 아직 저장하지 않는다."""
    _check_context_change(change)
    key = context_subject_key(change["media_id"], change["subject_id"])
    document = {name: change[name] for name in
                ("media_id", "subject_id", "version", "source_kind", "change_id", "event_time")}
    document.update({"_id": key, "deleted": change["op"] == "delete",
                     "context": {} if change["op"] == "delete" else dict(change["context"])})
    return document

import json

from django.conf import settings
from django.core.management.base import BaseCommand, CommandError
from pymongo.errors import PyMongoError

from ads.context_projection import context_subject_key, make_context_document
from ads.exporting import read_ndjson
from ads.mongo import get_db


class Command(BaseCommand):
    help = "정규화 문서를 미리보고 고유 인덱스와 현재 저장 상태를 확인합니다."

    def add_arguments(self, parser):
        parser.add_argument("--source")
        parser.add_argument("--media-id", default=settings.ADS_MEDIA_ID)
        parser.add_argument("--subject")

    def handle(self, *args, **options):
        try:
            media_id, subject = options["media_id"], options["subject"]
            context_subject_key(media_id, subject if subject is not None else "inspect")
            documents = None
            if options["source"]:
                documents = [make_context_document(row) for row in read_ndjson(options["source"])]
                if any(document["media_id"] != media_id for document in documents):
                    raise ValueError("입력 media_id와 --media-id가 다릅니다.")
            db = get_db()
            collection = db.player_context
            name = collection.create_index([("media_id", 1), ("subject_id", 1)],
                                           unique=True, name="media_subject_once")
            query = {"media_id": media_id}
            if subject is not None:
                query["_id"] = context_subject_key(media_id, subject)
            rows = list(collection.find(query).sort("_id", 1))
            result = {"database": db.name, "count": len(rows),
                      "index": collection.index_information()[name], "rows": rows,
                      "input_rows": None if documents is None else len(documents),
                      "document_preview": [] if documents is None else documents[:1]}
        except (OSError, TypeError, ValueError, KeyError, PyMongoError) as exc:
            raise CommandError(str(exc)) from exc
        self.stdout.write(json.dumps(result, ensure_ascii=False, sort_keys=True, indent=2))

def apply_context(change):
    document = make_context_document(change)
    key = document["_id"]
    collection = get_db().player_context
    # 먼저 읽은 상태가 여전히 맞을 때만 저장한다. 경쟁하면 다시 읽는다.
    for attempt in range(3):
        old = collection.find_one({"_id": key})
        if old is None:
            try:
                collection.insert_one(document)
                return "applied"
            except DuplicateKeyError:
                continue
        if old["version"] > document["version"]:
            return "stale"
        if old["version"] == document["version"]:
            if old["deleted"] == document["deleted"] and old["context"] == document["context"]:
                return "duplicate"
            raise ValueError("version_conflict: 같은 version의 상태가 다릅니다.")
        result = collection.replace_one({"_id": key, "version": old["version"]}, document)
        if result.matched_count == 1:
            return "applied"
    raise ValueError("concurrent_update: 최신 상태를 다시 읽고 재시도하세요.")

import json

from django.core.management.base import BaseCommand, CommandError
from pymongo.errors import PyMongoError

from ads.context_projection import apply_context, make_context_document, get_context
from ads.exporting import read_ndjson
from ads.mongo import get_db


class Command(BaseCommand):
    help = "정규화 파일의 지정 행을 최신 공개 context에 반영합니다."

    def add_arguments(self, parser):
        parser.add_argument("--source", required=True)
        parser.add_argument("--row", type=int, default=1)

    def handle(self, *args, **options):
        try:
            rows = read_ndjson(options["source"])
            documents = [make_context_document(row) for row in rows]
            number = options["row"]
            if not documents:
                raise ValueError("입력이 0행입니다. 반영할 대상이 없습니다.")
            if not 1 <= number <= len(documents):
                raise ValueError("--row는 1부터 입력 행 수 사이여야 합니다.")
            document = documents[number - 1]
            status = apply_context(rows[number - 1])
            database = get_db()
            stored = database.player_context.find_one({"_id": document["_id"]})
            result = {"database": database.name, "source": options["source"],
                      "row": number, "input_rows": len(rows), "status": status,
                      "key": document["_id"], "incoming_version": document["version"],
                      "stored_version": stored["version"], "stored_document": stored,
                      "context": get_context(document["media_id"], document["subject_id"])}
        except (OSError, ValueError, TypeError, KeyError, PyMongoError) as error:
            raise CommandError(str(error)) from error
        self.stdout.write(json.dumps(result, ensure_ascii=False))

            def observe(label, row, expect_conflict=False):
                before = collection.find_one({"_id": key})
                try:
                    status = apply_context(row)
                except ValueError as error:
                    if not expect_conflict or not str(error).startswith("version_conflict:"):
                        raise
                    status = "version_conflict"
                else:
                    if expect_conflict:
                        raise ValueError("동일 version의 다른 상태가 거절되지 않았습니다.")
                after = collection.find_one({"_id": key})
                observations.append({"step": label, "status": status,
                                     "incoming_version": row["version"],
                                     "stored_version": after["version"],
                                     "stored_coins": after["context"].get("coins"),
                                     "deleted": after["deleted"], "unchanged": before == after})

            newest = change(2, 4)
            observe("first", change(1, 3))
            observe("newer", newest)
            observe("repeat", newest)
            observe("older", change(1, 3))
            # 바뀐 내용으로 ID도 다시 계산해 version 충돌만 검증한다.
            observe("conflict", change(2, 99), expect_conflict=True)
            observe("delete", change(3, 0, "delete"))
            observe("older-after-delete", newest)
            result = {"database": database.name, "key": key, "steps": observations,
                      "stored_document": collection.find_one({"_id": key}),
                      "visible_context": get_context(media_id, subject)}
        except (OSError, ValueError, TypeError, KeyError, PyMongoError) as error:
            raise CommandError(str(error)) from error
        self.stdout.write(json.dumps(result, ensure_ascii=False))

def _read_context_changes(source):
    """같은 bytes에서 SHA와 검증한 정규화 행을 함께 얻는다."""
    payload = Path(source).read_bytes()
    rows = []
    for number, line in enumerate(payload.splitlines(), 1):
        if not line.strip():
            raise ValueError(f"빈 행 {number}")
        row = json.loads(line.decode("utf-8"))
        make_context_document(row)
        rows.append(row)
    return rows, hashlib.sha256(payload).hexdigest()


def _read_context_checkpoint(checkpoint, checksum, row_count):
    state = json.loads(Path(checkpoint).read_text(encoding="utf-8"))
    fields = {"schema_version", "source_sha256", "next_line"}
    if not isinstance(state, dict) or set(state) != fields:
        raise ValueError("checkpoint는 정해진 3개 필드만 받습니다.")
    if state["schema_version"] != "context-checkpoint/v1":
        raise ValueError("checkpoint 버전이 다릅니다.")
    digest = state["source_sha256"]
    if (not isinstance(digest, str) or len(digest) != 64
            or any(char not in "0123456789abcdef" for char in digest)):
        raise ValueError("checkpoint의 source_sha256이 잘못되었습니다.")
    if digest != checksum:
        raise ValueError("입력 파일이 바뀌었습니다. 새 checkpoint를 사용하세요.")
    position = state["next_line"]
    if type(position) is not int or not 0 <= position <= row_count:
        raise ValueError("checkpoint의 다음 행이 잘못되었습니다.")
    return position


def _save_context_checkpoint(checkpoint, checksum, position):
    """완성된 임시 파일로만 위치 기록을 교체한다."""
    target = Path(checkpoint)
    target.parent.mkdir(parents=True, exist_ok=True)
    state = {"schema_version": "context-checkpoint/v1",
             "source_sha256": checksum, "next_line": position}
    with tempfile.TemporaryDirectory(prefix=".checkpoint-", dir=target.parent) as folder:
        staged = Path(folder) / "state.json"
        staged.write_text(json.dumps(state, sort_keys=True), encoding="utf-8")
        os.replace(staged, target)
    return state

def project_context_file(source_path, checkpoint_path=None, limit=None):
    if limit is not None and (type(limit) is not int or limit < 1):
        raise ValueError("limit은 양수 정수입니다.")
    source = Path(source_path)
    if checkpoint_path is not None and not str(checkpoint_path).strip():
        raise ValueError("checkpoint 경로는 비어 있지 않아야 합니다.")
    checkpoint = Path(checkpoint_path) if checkpoint_path is not None else None
    if checkpoint is not None:
        if (source.resolve() == checkpoint.resolve()
                or (source.exists() and checkpoint.exists() and source.samefile(checkpoint))):
            raise ValueError("checkpoint는 원본 입력과 다른 파일이어야 합니다.")
    rows, checksum = _read_context_changes(source)
    position = 0
    state = None
    if checkpoint is not None and checkpoint.exists():
        position = _read_context_checkpoint(checkpoint, checksum, len(rows))
        state = {"schema_version": "context-checkpoint/v1",
                 "source_sha256": checksum, "next_line": position}
    start = position
    counts = {"applied": 0, "duplicate": 0, "stale": 0}
    stop = min(len(rows), position + limit) if limit is not None else len(rows)
    while position < stop:
        status = apply_context(rows[position])
        if status not in counts:
            raise ValueError("알 수 없는 반영 결과입니다.")
        next_position = position + 1
        if checkpoint is not None:
            state = _save_context_checkpoint(checkpoint, checksum, next_position)
        position = next_position
        counts[status] += 1
    # 빈 입력도 next_line=0을 저장한다. 기존 완료 기록은 덮어쓰지 않는다.
    if checkpoint is not None and state is None:
        state = _save_context_checkpoint(checkpoint, checksum, position)
    return {"processed": position - start, "applied": counts["applied"],
            "skipped": counts["duplicate"] + counts["stale"], "counts": counts,
            "start_line": start, "next_line": position, "rows": len(rows),
            "done": position == len(rows), "source_sha256": checksum,

from django.core.management.base import BaseCommand, CommandError
from pymongo.errors import PyMongoError

from ads.context_projection import project_context_file


class Command(BaseCommand):
    help = "같은 정규화 입력과 checkpoint로 중지·재개하며 전체 context를 반영합니다."

    def add_arguments(self, parser):
        parser.add_argument("--source", required=True)
        parser.add_argument("--checkpoint")
        parser.add_argument("--limit", type=int)

    def handle(self, *args, **options):
        try:
            result = project_context_file(options["source"], options["checkpoint"], options["limit"])
        except (OSError, ValueError, TypeError, KeyError, PyMongoError) as error:
            raise CommandError(str(error)) from error
        self.stdout.write(json.dumps(result, ensure_ascii=False))

def refresh_snapshot(source, contract_path, media_id="village-game"):
    context_subject_key(media_id, "refresh")
    rows, inspected = read_snapshot_package(source, contract_path)
    changes = [normalize_snapshot(row, media_id=media_id) for row in rows]
    counts = {"applied": 0, "duplicate": 0, "stale": 0}
    for change in changes:
        status = apply_context(change)
        if status not in counts:
            raise ValueError("알 수 없는 반영 결과입니다.")
        counts[status] += 1
    return {"source": str(source), "contract": str(contract_path), "rows": len(rows),
            "source_sha256": inspected["sha256"], "media_id": media_id,
            "counts": counts, "preview": changes[:1]}

from django.core.management.base import BaseCommand, CommandError
from pymongo.errors import PyMongoError

class Command(BaseCommand):
    help = "새 snapshot과 수신 계약을 대조한 뒤 공개 context를 갱신합니다."

    def add_arguments(self, parser):
        parser.add_argument("--source", required=True)
        parser.add_argument("--contract", required=True)
        parser.add_argument("--media-id", default=settings.ADS_MEDIA_ID)

    def handle(self, *args, **options):
        try:
            result = refresh_snapshot(options["source"], options["contract"],
                                      media_id=options["media_id"])
        except (OSError, ValueError, TypeError, KeyError, PyMongoError) as error:
            raise CommandError(str(error)) from error
        self.stdout.write(json.dumps(result, ensure_ascii=False, sort_keys=True, indent=2))

def request_context(player):
    media_id = settings.ADS_MEDIA_ID
    if not settings.ADS_MEDIA_KEY:
        raise AdsUnavailable("media_key_missing")
    subject = {"media_id": media_id, "subject_id": str(player.pk)}
    payload = {"subject": subject}
    headers = {"Content-Type": "application/json", "Accept": "application/json", "X-Media-ID": media_id, "X-Media-Key": settings.ADS_MEDIA_KEY}
    request = Request(
        settings.ADS_BASE_URL.rstrip("/") + "/api/media/context/",
        data=json.dumps(payload, allow_nan=False).encode("utf-8"),
        headers=headers,
        method="POST",
    )
    try:
        with urlopen(request, timeout=1.5) as response:
            data = json.load(response)
        if (not isinstance(data, dict) or data.get("subject") != subject
                or not isinstance(data.get("context"), dict)):
            raise ValueError("invalid_context_response")
        context = data["context"]
        if context:
            if (set(context) != {"player_id", "room_id", "coins", "version", "updated_at"}
                    or context["player_id"] != subject["subject_id"]
                    or not isinstance(context["room_id"], str)
                    or type(context["coins"]) is not int
                    or type(context["version"]) is not int or context["version"] < 0
                    or not isinstance(context["updated_at"], str)):
                raise ValueError("invalid_public_context")
            instant = datetime.fromisoformat(context["updated_at"])
            if instant.tzinfo is None or instant.utcoffset() is None:
                raise ValueError("context_time_requires_timezone")
        return dict(context)
    except (HTTPError, URLError, TimeoutError, OSError, ValueError, KeyError, TypeError) as error:
        raise AdsUnavailable("ads_unavailable") from error

def request_decision(player, slot_id):
    media_id = settings.ADS_MEDIA_ID
    if not settings.ADS_MEDIA_KEY:
        raise AdsUnavailable("media_key_missing")
    payload = {
        "media_id": media_id,
        "subject": {"media_id": media_id, "subject_id": str(player.pk)},
        "slot_id": slot_id,
        "context": request_context(player),
    }
    request = Request(
        settings.ADS_BASE_URL.rstrip("/") + "/api/media/decision/",
        data=json.dumps(payload).encode("utf-8"),
        headers={"Content-Type": "application/json", "Accept": "application/json",
                 "X-Media-ID": media_id, "X-Media-Key": settings.ADS_MEDIA_KEY},
        method="POST",
    )
    try:
        with urlopen(request, timeout=1.5) as response:
            data = json.load(response)
        if not isinstance(data, dict):
            raise ValueError("invalid_response")
        if data.get("empty") is True or data.get("ad", "selected") is None:
            return {**data, "ad": None}
        if data.get("slot_id") != slot_id:
            raise ValueError("slot_mismatch")
        return {**data, "body": data.get("body", ""),
                "bid_amount": data.get("bid_amount", data.get("bid_units"))}
    except (HTTPError, URLError, TimeoutError, OSError, ValueError, KeyError, TypeError) as error:
        raise AdsUnavailable("ads_unavailable") from error

from django.conf import settings
from django.http import JsonResponse
from django.views.decorators.http import require_GET

from game.models import Player
from game.ad_gateway import AdsUnavailable, request_context


@require_GET
def ad_context(request):
    if not request.user.is_authenticated:
        return JsonResponse({"error": "login_required"}, status=401)
    player = Player.objects.filter(user=request.user).first()
    if player is None:
        return JsonResponse({"error": "player_missing"}, status=404)
    try:
        response = JsonResponse({"subject": {"media_id": settings.ADS_MEDIA_ID,
                                              "subject_id": str(player.pk)},
                                 "context": request_context(player)})
        response["Cache-Control"] = "no-store"
        return response
    except AdsUnavailable:
        return JsonResponse({"error": "ads_unavailable"}, status=503)

import json

from django.conf import settings
from django.core.management.base import BaseCommand, CommandError
from pymongo.errors import PyMongoError

from ads.context_projection import context_subject_key, get_context
from ads.mongo import get_db


class Command(BaseCommand):
    help = "선택 당시 context와 현재 공개 복사본을 읽기만 하여 비교합니다."

    def add_arguments(self, parser):
        choice = parser.add_mutually_exclusive_group(required=True)
        choice.add_argument("--decision-id")
        choice.add_argument("--subject")
        parser.add_argument("--latest", action="store_true")
        parser.add_argument("--media-id", default=settings.ADS_MEDIA_ID)

    def handle(self, *args, **options):
        try:
            media_id = options["media_id"]
            context_subject_key(media_id, options["subject"] or "inspection")
            query = {"subject.media_id": media_id}
            if options["decision_id"] is not None:
                if options["latest"]:
                    raise ValueError("--decision-id와 --latest를 함께 지정하지 않습니다.")
                if not options["decision_id"].strip():
                    raise ValueError("decision_id는 비어 있지 않아야 합니다.")
                query["_id"] = options["decision_id"]
            else:
                if not options["latest"]:
                    raise ValueError("--subject 조회에는 --latest를 함께 지정하세요.")
                query["subject.subject_id"] = options["subject"]
            projection = {"_id": 1, "subject": 1, "context": 1, "selected_at": 1}
            row = get_db().decisions.find_one(query, projection,
                                              sort=[("selected_at", -1), ("_id", -1)])
            if row is None:
                raise ValueError("선택 기록이 없습니다. empty 응답은 decision을 저장하지 않습니다.")
            subject = row["subject"]
            saved = dict(row["context"])
            current = get_context(subject["media_id"], subject["subject_id"])
            names = sorted(set(saved) | set(current))
            differences = {name: {"decision": saved.get(name), "current": current.get(name)}
                           for name in names if saved.get(name) != current.get(name)
                           or (name in saved) != (name in current)}
            instant = row.get("selected_at")
            result = {"decision_id": row["_id"], "subject": subject,
                      "selected_at": instant.isoformat() if hasattr(instant, "isoformat") else instant,
                      "context": saved, "current_context": current,
                      "same_context": saved == current, "differences": differences}
        except (ValueError, TypeError, KeyError, PyMongoError) as error:
            raise CommandError(str(error)) from error
        self.stdout.write(json.dumps(result, ensure_ascii=False, indent=2, sort_keys=True))

실제

def _read_handoff_context(media_id):
    rows = list(get_db().player_context.find({"media_id": media_id}).sort("_id", 1))
    fields = {"_id", "media_id", "subject_id", "version", "deleted", "context",
              "source_kind", "change_id", "event_time"}
    for row in rows:
        if set(row) != fields or type(row["deleted"]) is not bool:
            raise ValueError("저장 context의 공개 문서 계약이 다릅니다.")
        change = {"schema_version": "player-context-change/v1",
                  "op": "delete" if row["deleted"] else "upsert",
                  **{name: row[name] for name in fields - {"_id", "deleted"}}}
        if make_context_document(change) != row:
            raise ValueError("저장 context의 대상 키와 내용이 다릅니다.")
    return rows

import json

from django.conf import settings
from django.core.management.base import BaseCommand, CommandError
from pymongo.errors import PyMongoError

from ads.context_projection import export_context_handoff


class Command(BaseCommand):
    help = "검증한 입력·완료 기록·현재 공개 복사본을 새 인계 폴더로 묶습니다."

    def add_arguments(self, parser):
        parser.add_argument("--source", required=True)
        parser.add_argument("--checkpoint", required=True)
        parser.add_argument("--output-dir", required=True)
        parser.add_argument("--media-id", default=settings.ADS_MEDIA_ID)

    def handle(self, *args, **options):
        try:
            result = export_context_handoff(options["source"], options["checkpoint"],
                                            options["output_dir"], options["media_id"])
        except (OSError, ValueError, TypeError, KeyError, PyMongoError) as error:
            raise CommandError(str(error)) from error
        self.stdout.write(json.dumps(result, ensure_ascii=False))

잘못된 함수는 화면의 함수 전체와 대조해 수정한다. 원본·완료 기록·기존 인계 폴더·MongoDB를 삭제하지 않는다. 새 출력에는 존재하지 않는 경로를 사용한다.
재실행 증거:
같은 광고 작업 폴더에서 새 출력 경로로 다시 실행한다.
python ad_config/manage.py export_context_handoff --source data/exports/player-context-changes-day25-after.ndjson --checkpoint data/checkpoints/player-context-day25-after.json --output-dir data/handoff/day25-run2
반환한 입력 SHA·현재 복사본 SHA·경로·완료 기록을 읽고 6단계 소비 명령도 run2의 사본 경로로 실행한다.
인계할 네 파일을 직접 설명한다
자신의 manifest에서 input_path·input_sha256·checkpoint.next_line·context_path·context_sha256를 찾아 역할별로 적는다. fixture가 들어 있어 input_rows와 context_rows가 다를 때 이를 오류로 단정하면 안 되는 이유를 source_kind 집계로 설명한다. 다른 source SHA의 완료 위치를 인계하면 왜 거절해야 하는지 실제 출력으로 답한다.
8교시 완료 기준
실제 export_context_handoff의 네 빈칸을 완성했다. 6교시 새 입력의 완료 기록을 만들고, 입력 사본·기록·현재 공개 복사본·manifest 네 파일을 새 폴더로 인계했다. 기존 폴더와 미완료 기록의 거절, 사본의 실제 소비, 원본·기존 기록 보존을 확인했다. 안정 artifact ID와 schema·출처·두 지문을 다음 소비자에게 넘긴다.
고급 선택:
현재 MongoDB 문서들을 읽는 동안 다른 작성자가 갱신하면 여러 문서의 동일 시점 snapshot을 보장하지 않는 이유를 설명한다. 이 수업은 동결 입력과 한 작성자의 인계 모델이며, 다음 입력이 추가되더라도 기존 인계 폴더를 덮어쓰지 않는다.
