"""Day 22: choose a database and the practice collection, then ping."""
import os
from pathlib import Path

from dotenv import load_dotenv
from pymongo import MongoClient


def main():
    root = Path(__file__).resolve().parents[2]
    load_dotenv(root / "ads.env", override=True)
    load_dotenv(root / ".env", override=True)
    with MongoClient(os.environ["MONGO_URI"], serverSelectionTimeoutMS=3000,
                     connectTimeoutMS=3000) as client:
        database = client[os.environ.get("MONGO_DB", "village_ads")]
        collection = database["practice_campaigns"]
        print("database", database.name)
        print("collection", collection.name)
        print("server", client.admin.command("ping")["ok"])
    print("closed")


if __name__ == "__main__":
    main()
